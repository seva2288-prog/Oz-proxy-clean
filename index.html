state.futureMatches = []; state.refreshTimer = null; state.refreshCountdown = 300; // --- СТАТУС МАТЧА --- function getMatchStatus(matchDate, matchGoals) { const now = new Date(); const matchTime = new Date(matchDate); const diffMs = matchTime - now; const diffMin = Math.floor(diffMs / 60000); if (matchGoals && matchGoals.home !== undefined && matchGoals.away !== undefined) { return { status: 'finished', label: `✅ Завершён ${matchGoals.home}:${matchGoals.away}`, emoji: '✅', color: '#555' }; } if (diffMin <= 0 && diffMin > -5) { return { status: 'live', label: '🔴 Идёт', emoji: '🔴', color: '#e74c3c' }; } if (diffMin < -5) { return { status: 'finished', label: '⏳ Завершён (результат неизвестен)', emoji: '⏳', color: '#555' }; } const hours = Math.floor(diffMin / 60); const minutes = diffMin % 60; let label = ''; if (hours > 24) { const days = Math.floor(hours / 24); label = `📅 через ${days} дн ${hours % 24} ч`; } else if (hours > 0) { label = `🟢 через ${hours} ч ${minutes} мин`; } else { label = `🟢 через ${minutes} мин`; } return { status: 'future', label: label, emoji: '🟢', color: '#2ecc71', hoursUntil: hours, minutesUntil: minutes }; } // --- ПРИОРИТЕТ --- function calculatePriority(confidence, hoursUntil) { const timeFactor = 1 / (Math.max(hoursUntil, 0.1) + 1); return Math.round(((confidence / 100) * timeFactor * 100) * 100) / 100; } // --- ФИЛЬТРЫ --- function filterMatches(matches, filters) { const now = new Date(); return matches.filter(m => { const matchDate = new Date(m.date); const diffMs = matchDate - now; const diffHours = diffMs / 3600000; if (filters.time === '6h' && (diffHours < 0 || diffHours > 6)) return false; if (filters.time === 'today' && matchDate.getDate() !== now.getDate()) return false; if (filters.time === 'tomorrow') { const tomorrow = new Date(now); tomorrow.setDate(tomorrow.getDate() + 1); if (matchDate.getDate() !== tomorrow.getDate()) return false; } if (filters.time === 'week') { const weekLater = new Date(now); weekLater.setDate(weekLater.getDate() + 7); if (matchDate > weekLater || matchDate < now) return false; } if (m.confidence < filters.confidence) return false; if (filters.total !== 'all') { const total = m.total || 0; if (filters.total === 'over15' && total <= 1.5) return false; if (filters.total === 'over25' && total <= 2.5) return false; if (filters.total === 'over35' && total <= 3.5) return false; } return true; }); } // --- СОРТИРОВКА ПО ПРИОРИТЕТУ --- function sortMatchesByPriority(matches) { return [...matches].sort((a, b) => { const statusOrder = { live: 0, future: 1, finished: 2 }; const statusA = a.statusData?.status || 'future'; const statusB = b.statusData?.status || 'future'; if (statusOrder[statusA] !== statusOrder[statusB]) { return statusOrder[statusA] - statusOrder[statusB]; } return (b.priority || 0) - (a.priority || 0); }); } console.log('✅ ЧАСТЬ 1/4 — Стейт, статусы, приоритет, фильтры — ЗАГРУЖЕНА');
